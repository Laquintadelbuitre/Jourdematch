/** * MATCHDAY LIVE — backend Google Apps Script * ------------------------------------------------- * A coller dans l'éditeur Apps Script lié à votre Google Sheet * (Extensions > Apps Script, depuis le Sheet dédié à cet outil). * * 1) Exécutez une fois la fonction setup() (menu Exécuter > setup) * pour créer/mettre à jour les onglets et les en-têtes. * (Sans danger à relancer plusieurs fois : ça ne touche jamais aux * données déjà saisies, seulement à la structure des onglets.) * 2) Déployez : Déployer > Nouveau déploiement > Type "Application Web" * - Exécuter en tant que : Moi * - Qui a accès : Tout le monde * 3) Copiez l'URL "/exec" fournie et collez-la dans API_URL en haut * de index.html. */ const MOT_DE_PASSE_COACH = '0501'; const SHEET_CALENDRIER = 'Calendrier'; const SHEET_EFFECTIF = 'Effectif'; const SHEET_EQUIPES = 'Equipes'; const SHEET_CONVOCATIONS = 'Convocations'; const SHEET_COMPOSITIONS = 'Compositions'; const SHEET_EVENEMENTS = 'Evenements'; /** * Crée les onglets manquants et ajoute les colonnes manquantes sur les * onglets déjà existants (sans jamais supprimer de données). À exécuter * à chaque fois qu'une mise à jour du script ajoute de nouveaux champs. */ function setup() { const ss = SpreadsheetApp.getActiveSpreadsheet(); const sheets = { [SHEET_CALENDRIER]: ['matchId', 'date', 'heureDebut', 'equipe', 'adversaire', 'lieu'], [SHEET_EFFECTIF]: ['joueur'], [SHEET_EQUIPES]: ['equipe'], [SHEET_CONVOCATIONS]: ['matchId', 'joueur'], [SHEET_COMPOSITIONS]: ['matchId', 'joueur', 'poste'], [SHEET_EVENEMENTS]: ['eventId', 'matchId', 'horodatage', 'minute', 'type', 'buteur', 'passeur', 'statut', 'saisiPar'], }; Object.keys(sheets).forEach(name => { let sh = ss.getSheetByName(name); if (!sh) sh = ss.insertSheet(name); if (sh.getLastRow() === 0) { sh.appendRow(sheets[name]); sh.setFrozenRows(1); return; } // Migration douce : ajoute en fin de ligne les colonnes manquantes // sans toucher aux colonnes/données existantes. const headersActuels = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]; const manquantes = sheets[name].filter(h => headersActuels.indexOf(h) === -1); if (manquantes.length) { sh.getRange(1, headersActuels.length + 1, 1, manquantes.length).setValues([manquantes]); } }); } /* ---------------------------- Helpers ---------------------------- */ function getSheet_(name) { return SpreadsheetApp.getActiveSpreadsheet().getSheetByName(name); } /** Convertit les valeurs Date renvoyées par Sheets en texte lisible pour les champs concernés. */ function normaliserValeur_(header, value) { if (value instanceof Date) { if (header === 'date') return Utilities.formatDate(value, Session.getScriptTimeZone(), 'yyyy-MM-dd'); if (header === 'heureDebut') return Utilities.formatDate(value, Session.getScriptTimeZone(), 'HH:mm'); if (header === 'horodatage') return value; // reste un Date, utile pour les comparaisons } return value; } function sheetToObjects_(name) { const sh = getSheet_(name); const values = sh.getDataRange().getValues(); const headers = values.shift(); return values .filter(row => row.some(cell => cell !== '' && cell !== null)) .map(row => { const obj = {}; headers.forEach((h, i) => (obj[h] = normaliserValeur_(h, row[i]))); return obj; }); } function appendObject_(name, obj) { const sh = getSheet_(name); const headers = sh.getRange(1, 1, 1, sh.getLastColumn()).getValues()[0]; const row = headers.map(h => (obj[h] !== undefined ? obj[h] : '')); const nouvelleLigne = sh.getLastRow() + 1; const range = sh.getRange(nouvelleLigne, 1, 1, row.length); // Force le texte brut sur toute la ligne : empêche Sheets de convertir // automatiquement une valeur comme "22:00" ou "2026-08-25" en heure/date interne. range.setNumberFormat('@'); range.setValues([row]); } function jsonOut_(data) { return ContentService.createTextOutput(JSON.stringify(data)).setMimeType( ContentService.MimeType.JSON ); } function newId_(prefix) { return prefix + '_' + Utilities.getUuid().slice(0, 8); } function checkPwd_(pwd) { return pwd === MOT_DE_PASSE_COACH; } /* ------------------------------ GET ------------------------------- */ function doGet(e) { const action = e.parameter.action; try { if (action === 'calendrier') { return jsonOut_({ ok: true, calendrier: sheetToObjects_(SHEET_CALENDRIER) }); } if (action === 'effectif') { return jsonOut_({ ok: true, effectif: sheetToObjects_(SHEET_EFFECTIF).map(r => r.joueur) }); } if (action === 'equipes') { return jsonOut_({ ok: true, equipes: sheetToObjects_(SHEET_EQUIPES).map(r => r.equipe) }); } if (action === 'match') { return jsonOut_({ ok: true, match: getMatchState_(e.parameter.matchId) }); } return jsonOut_({ ok: false, error: 'Action GET inconnue.' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } /** Construit l'état complet d'un match : infos, convocations, score, événements actifs. */ function getMatchState_(matchId) { const calendrier = sheetToObjects_(SHEET_CALENDRIER).find(m => m.matchId === matchId); const convocations = sheetToObjects_(SHEET_CONVOCATIONS) .filter(c => c.matchId === matchId) .map(c => c.joueur); const composition = sheetToObjects_(SHEET_COMPOSITIONS).filter(c => c.matchId === matchId); const evenements = sheetToObjects_(SHEET_EVENEMENTS).filter(ev => ev.matchId === matchId); const valides = evenements.filter(ev => ev.statut === 'valide'); const scoreNous = valides.filter(ev => ev.type === 'but').length; const scoreAdverse = valides.filter(ev => ev.type === 'but_adverse').length; const derniereAction = evenements.length ? evenements.reduce((a, b) => (new Date(a.horodatage) > new Date(b.horodatage) ? a : b)) : null; return { info: calendrier || null, convocations, composition, evenements: valides, scoreNous, scoreAdverse, derniereAction: derniereAction && derniereAction.statut === 'valide' ? derniereAction : null, }; } /* ------------------------------ POST ------------------------------ */ function doPost(e) { let body; try { body = JSON.parse(e.postData.contents); } catch (err) { return jsonOut_({ ok: false, error: 'Corps de requête invalide.' }); } const action = body.action; try { // Actions réservées au coach if (['creerMatch', 'setConvocations', 'setComposition', 'setEffectif', 'setEquipes', 'syncConvocations'].includes(action)) { if (!checkPwd_(body.pwd)) return jsonOut_({ ok: false, error: 'Mot de passe incorrect.' }); } if (action === 'creerMatch') return creerMatch_(body); if (action === 'setConvocations') return setConvocations_(body); if (action === 'syncConvocations') return syncConvocations_(body); if (action === 'setComposition') return setComposition_(body); if (action === 'setEffectif') return setEffectif_(body); if (action === 'setEquipes') return setEquipes_(body); // Actions ouvertes (parents + coach) if (action === 'enregistrerBut') return enregistrerBut_(body); if (action === 'enregistrerPasse') return enregistrerPasse_(body); if (action === 'butAdverse') return butAdverse_(body); if (action === 'annulerDerniere') return annulerDerniere_(body); return jsonOut_({ ok: false, error: 'Action POST inconnue.' }); } catch (err) { return jsonOut_({ ok: false, error: String(err) }); } } function creerMatch_(body) { const matchId = newId_('match'); appendObject_(SHEET_CALENDRIER, { matchId, date: body.date, heureDebut: body.heureDebut, equipe: body.equipe || '', adversaire: body.adversaire, lieu: body.lieu || '', }); return jsonOut_({ ok: true, matchId }); } function setEffectif_(body) { const sh = getSheet_(SHEET_EFFECTIF); const nbLignesExistantes = sh.getLastRow() - 1; if (nbLignesExistantes > 0) { sh.getRange(2, 1, nbLignesExistantes, 1).clearContent(); } const joueurs = body.joueurs || []; if (joueurs.length) { sh.getRange(2, 1, joueurs.length, 1).setNumberFormat('@').setValues(joueurs.map(j => [j])); } return jsonOut_({ ok: true }); } function setEquipes_(body) { const sh = getSheet_(SHEET_EQUIPES); const nbLignesExistantes = sh.getLastRow() - 1; if (nbLignesExistantes > 0) { sh.getRange(2, 1, nbLignesExistantes, 1).clearContent(); } const equipes = body.equipes || []; if (equipes.length) { sh.getRange(2, 1, equipes.length, 1).setNumberFormat('@').setValues(equipes.map(eq => [eq])); } return jsonOut_({ ok: true }); } function setConvocations_(body) { ecrireConvocations_(body.matchId, body.joueurs || []); return jsonOut_({ ok: true }); } /** Écrit la liste des convoqués pour un match (remplace les convocations existantes). */ function ecrireConvocations_(matchId, joueurs) { const sh = getSheet_(SHEET_CONVOCATIONS); const values = sh.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (values[i][0] === matchId) sh.deleteRow(i + 1); } joueurs.forEach(j => sh.appendRow([matchId, j])); } /** * Reçoit une convocation générée depuis un autre outil (ex. convoc-LF) et * l'associe au bon match dans Calendrier : retrouve un match existant pour * la même équipe + la même date, sinon en crée un. Met aussi à jour * adversaire/heure/lieu si transmis. */ function syncConvocations_(body) { const existant = sheetToObjects_(SHEET_CALENDRIER) .find(m => m.equipe === body.equipe && m.date === body.date); let matchId; if (existant) { matchId = existant.matchId; mettreAJourMatch_(matchId, body); } else { matchId = newId_('match'); appendObject_(SHEET_CALENDRIER, { matchId, date: body.date || '', heureDebut: body.heureDebut || '', equipe: body.equipe || '', adversaire: body.adversaire || '', lieu: body.lieu || '', }); } ecrireConvocations_(matchId, body.joueurs || []); return jsonOut_({ ok: true, matchId }); } /** Met à jour adversaire/heureDebut/lieu d'un match existant (uniquement les champs fournis et non vides). */ function mettreAJourMatch_(matchId, body) { const sh = getSheet_(SHEET_CALENDRIER); const values = sh.getDataRange().getValues(); const headers = values[0]; const idCol = headers.indexOf('matchId'); for (let i = 1; i < values.length; i++) { if (values[i][idCol] === matchId) { ['adversaire', 'heureDebut', 'lieu'].forEach(champ => { if (body[champ]) { const col = headers.indexOf(champ); if (col > -1) sh.getRange(i + 1, col + 1).setNumberFormat('@').setValue(body[champ]); } }); break; } } } function setComposition_(body) { const sh = getSheet_(SHEET_COMPOSITIONS); const values = sh.getDataRange().getValues(); for (let i = values.length - 1; i >= 1; i--) { if (values[i][0] === body.matchId) sh.deleteRow(i + 1); } (body.composition || []).forEach(c => sh.appendRow([body.matchId, c.joueur, c.poste || ''])); return jsonOut_({ ok: true }); } function minuteDepuisDebut_(matchId, horodatage) { const match = sheetToObjects_(SHEET_CALENDRIER).find(m => m.matchId === matchId); if (!match || !match.date || !match.heureDebut) return ''; try { const debut = new Date(`${match.date}T${match.heureDebut}:00`); const diffMin = Math.round((horodatage - debut) / 60000); return diffMin >= 0 ? diffMin : ''; } catch (err) { return ''; } } function enregistrerBut_(body) { const eventId = newId_('evt'); const horodatage = new Date(); appendObject_(SHEET_EVENEMENTS, { eventId, matchId: body.matchId, horodatage: horodatage.toISOString(), minute: minuteDepuisDebut_(body.matchId, horodatage), type: 'but', buteur: body.buteur, passeur: '', statut: 'valide', saisiPar: body.role || 'parent', }); return jsonOut_({ ok: true, eventId, match: getMatchState_(body.matchId) }); } function enregistrerPasse_(body) { const sh = getSheet_(SHEET_EVENEMENTS); const values = sh.getDataRange().getValues(); const headers = values[0]; const eventIdCol = headers.indexOf('eventId'); const passeurCol = headers.indexOf('passeur'); for (let i = 1; i < values.length; i++) { if (values[i][eventIdCol] === body.eventId) { sh.getRange(i + 1, passeurCol + 1).setValue(body.passeur); break; } } return jsonOut_({ ok: true, match: getMatchState_(body.matchId) }); } function butAdverse_(body) { const eventId = newId_('evt'); const horodatage = new Date(); appendObject_(SHEET_EVENEMENTS, { eventId, matchId: body.matchId, horodatage: horodatage.toISOString(), minute: minuteDepuisDebut_(body.matchId, horodatage), type: 'but_adverse', buteur: '', passeur: '', statut: 'valide', saisiPar: body.role || 'parent', }); return jsonOut_({ ok: true, match: getMatchState_(body.matchId) }); } /** N'annule que la toute dernière action (par horodatage) du match, quel que soit son type. */ function annulerDerniere_(body) { const sh = getSheet_(SHEET_EVENEMENTS); const values = sh.getDataRange().getValues(); const headers = values[0]; const matchIdCol = headers.indexOf('matchId'); const horodatageCol = headers.indexOf('horodatage'); const statutCol = headers.indexOf('statut'); let lastRowIndex = -1; let lastDate = null; for (let i = 1; i < values.length; i++) { if (values[i][matchIdCol] === body.matchId && values[i][statutCol] === 'valide') { const d = new Date(values[i][horodatageCol]); if (!lastDate || d > lastDate) { lastDate = d; lastRowIndex = i; } } } if (lastRowIndex === -1) return jsonOut_({ ok: false, error: 'Aucune action à annuler.' }); sh.getRange(lastRowIndex + 1, statutCol + 1).setValue('annule'); return jsonOut_({ ok: true, match: getMatchState_(body.matchId) }); }
